## 浅拷贝

```javascript
function shallowClone(target) {
  // 边界处理：非对象/数组直接返回（原始类型按值传递）
  if (typeof target !== "object" || target === null) {
    return target;
  }

  // 判断是数组还是对象，初始化新容器
  const clone = Array.isArray(target) ? [] : {};

  // 遍历表层属性，复制（嵌套对象/数组仍引用原内存）
  for (const key in target) {
    if (target.hasOwnProperty(key)) { // 只复制自身属性（排除原型链）
      clone[key] = target[key];
    }
  }

  return clone;
}
```
