## 数组去重

### Set + 扩展运算符（推荐）

```js
// 基本类型数组去重（数字、字符串、布尔值等）
const arr = [1, 2, 2, 3, 3, 3, 'a', 'a', true, true];
const uniqueArr = [...new Set(arr)];
console.log(uniqueArr); // 输出：[1, 2, 3, 'a', true]

// 也可以用 Array.from 转换 Set 为数组（效果一样）
const uniqueArr2 = Array.from(new Set(arr));
console.log(uniqueArr2); // 输出：[1, 2, 3, 'a', true]
```

- **无法去重引用类型（如对象、数组）**，因为 Set 判重是基于 “严格相等 ===”，不同引用的对象即使内容相同也会被认为是不同元素
- - 对 NaN 友好：**Set 会把多个 NaN 视为同一个**（而 NaN === NaN 是 false）

### 手写：利用 indexOf/includes

**1. indexOf 版**

```js
function unique(arr) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    // 判断当前元素是否已在结果数组中（indexOf 返回 -1 表示不存在）
    if (result.indexOf(arr[i]) === -1) {
      result.push(arr[i]);
    }
  }
  return result;
}

// 测试
const arr = [1, 2, 2, 3, 3];
console.log(unique(arr)); // [1, 2, 3]
```

<br/>

**2. includes 版**

```js
function unique(arr) {
  const result = [];
  for (const item of arr) {
    // includes 直接判断是否存在，比 indexOf 更易读
    if (!result.includes(item)) {
      result.push(item);
    }
  }
  return result;
}
```

### 大数据量数组

**1. Map 版**

```js
function unique(arr) {
  const map = new Map();
  const result = [];
  for (const item of arr) {
    // Map 的 key 不会重复，判断是否已存在
    if (!map.has(item)) {
      map.set(item, true); // 标记为已存在
      result.push(item);
    }
  }
  return result;
}

// 测试
const arr = [1, 2, 2, 3, 3, 3];
console.log(unique(arr)); // [1, 2, 3]
```

<br/>

**2. 对象版**

```js
function unique(arr) {
  const obj = {};
  const result = [];
  for (const item of arr) {
    // 用元素作为对象的 key，避免重复
    // 注意：需拼接类型，避免 1 和 '1' 被视为同一个
    const key = typeof item + item;
    if (!obj[key]) {
      obj[key] = true;
      result.push(item);
    }
  }
  return result;
}

// 测试（区分数字和字符串）
const arr = [1, '1', 1, 2, '2'];
console.log(unique(arr)); // [1, '1', 2, '2']
```
