## 数组拍平

### flat() 方法

```js
// 一维嵌套数组
const arr1 = [1, [2, 3], 4];
console.log(arr1.flat()); // 输出：[1, 2, 3, 4]

// 多维嵌套数组
const arr2 = [1, [2, [3, [4]]]];
// 参数：拍平的深度，Infinity 表示拍平所有层级
console.log(arr2.flat(Infinity)); // 输出：[1, 2, 3, 4]

// 处理空元素（flat() 会自动移除数组中的空项）
const arr3 = [1, , [2, , 3]];
console.log(arr3.flat()); // 输出：[1, 2, 3]
```

> **flat() 会自动移除数组中的空项**

### 手写实现

**遍历数组，判断每个元素是否是数组，若是则递归处理，否则直接收集**

<br/>

**1. 基础递归版（拍平所有层级）**
```js
function flatten(arr) {
  let result = []; // 存储最终的一维数组
  
  // 遍历原数组的每个元素
  for (let item of arr) {
    // 判断当前元素是否是数组（注意：Array.isArray 是最可靠的判断方式）
    if (Array.isArray(item)) {
      // 若是数组，递归调用flatten，再把结果合并到result中
      result = result.concat(flatten(item));
    } else {
      // 若不是数组，直接添加到result
      result.push(item);
    }
  }
  
  return result;
}

// 测试
const arr = [1, [2, [3, [4]]], 5];
console.log(flatten(arr)); // 输出：[1, 2, 3, 4, 5]
```

<br/>

**2. 带深度控制的递归版（更贴近原生 flat）**

```js
function flattenWithDepth(arr, depth = 1) {
  let result = [];
  
  for (let item of arr) {
    // 只有当元素是数组且深度>0时，才递归拍平（深度-1）
    if (Array.isArray(item) && depth > 0) {
      result = result.concat(flattenWithDepth(item, depth - 1));
    } else {
      result.push(item);
    }
  }
  
  return result;
}

// 测试
const arr = [1, [2, [3, [4]]], 5];
console.log(flattenWithDepth(arr, 2)); // 输出：[1, 2, 3, [4], 5]（只拍平2层）
console.log(flattenWithDepth(arr, Infinity)); // 输出：[1, 2, 3, 4, 5]（拍平所有层）
```

3. 循环实现，避免栈溢出
```js
function flattenLoop(arr) {
  let result = [...arr]; // 复制原数组，避免修改原数据
  
  // 只要数组中还有嵌套的数组，就继续处理
  while (result.some(item => Array.isArray(item))) {
    // 展开一层数组（concat会自动展开数组参数）
    result = [].concat(...result);
  }
  
  return result;
}

// 测试
const arr = [1, [2, [3, [4]]], 5];
console.log(flattenLoop(arr)); // 输出：[1, 2, 3, 4, 5]
```
