## 实现深拷贝/浅拷贝，处理循环引用

### 浅拷贝

```js
function shallowClone(target) {
  // 边界处理：非对象/数组直接返回（原始类型按值传递）
  if (typeof target !== "object" || target === null) {
    return target;
  }

  // 判断是数组还是对象，初始化新容器
  const cloneTarget = Array.isArray(target) ? [] : {};

  // 遍历表层属性，复制（嵌套对象/数组仍引用原内存）
  for (const key in target) {
    if (target.hasOwnProperty(key)) { // 只复制自身属性（排除原型链）
      cloneTarget[key] = target[key];
    }
  }

  return cloneTarget;
}

// 测试浅拷贝
const obj1 = { a: 1, b: { c: 2 } };
const obj1Clone = shallowClone(obj1);
obj1Clone.b.c = 3;
console.log(obj1.b.c); // 3（嵌套对象共享内存，修改拷贝影响原对象）
```

### 支持循环引用的深拷贝

```js
/**
 * 深拷贝：递归复制所有层级，处理循环引用
 * @param {any} target - 要拷贝的目标
 * @param {WeakMap} cache - 缓存已拷贝的对象，处理循环引用
 * @returns 拷贝后的新对象/数组
 */
function deepClone(target, cache = new WeakMap()) {
  // 1. 边界处理：原始类型直接返回（按值传递）
  if (typeof target !== "object" || target === null) {
    return target;
  }

  // 2. 处理循环引用：如果已拷贝过该对象，直接返回缓存的副本
  if (cache.has(target)) {
    return cache.get(target);
  }

  // 3. 初始化新容器（区分数组/对象/特殊对象）
  let cloneTarget;
  // 处理数组
  if (Array.isArray(target)) {
    cloneTarget = [];
  }
  // 处理日期
  else if (target instanceof Date) {
    cloneTarget = new Date(target);
  }
  // 处理正则
  else if (target instanceof RegExp) {
    cloneTarget = new RegExp(target.source, target.flags);
  }
  // 处理普通对象
  else {
    cloneTarget = {};
  }

  // 4. 缓存当前对象（关键：解决循环引用）
  cache.set(target, cloneTarget);

  // 5. 递归拷贝属性（遍历自身可枚举属性，包括Symbol）
  const keys = [...Object.keys(target), ...Object.getOwnPropertySymbols(target)];
  for (const key of keys) {
    // 递归拷贝每个属性的值
    cloneTarget[key] = deepClone(target[key], cache);
  }

  return cloneTarget;
}

// 测试循环引用的深拷贝
const obj2 = { a: 1, b: { c: 2 } };
obj2.self = obj2; // 循环引用：obj2.self指向自身

const obj2Clone = deepClone(obj2);
obj2Clone.b.c = 3;
obj2Clone.self.a = 100;

console.log(obj2.b.c); // 2（嵌套对象独立，修改拷贝不影响原对象）
console.log(obj2.self.a); // 1（循环引用的属性也独立）
console.log(obj2Clone.self === obj2Clone); // true（拷贝后的循环引用指向自身副本）
```

> **为什么用 WeakMap 做缓存？**
    >
    > WeakMap 的键是「弱引用」，不会阻止垃圾回收（拷贝完成后，缓存对象可被自动清理，避免内存泄漏）；
如果用普通 Map，键是强引用，缓存对象会一直占用内存，可能导致内存溢出。
