## 对象数组去重

<br/>

### 按单个属性去重

##### 1. Map + filter（推荐）

```js
function uniqueObjArrByKey(arr, key) {
  // 校验参数，避免报错
  if (!Array.isArray(arr) || !key) return arr;
  
  const map = new Map();
  // filter 过滤：只保留首次出现的元素
  return arr.filter(item => {
    // 确保对象有该属性，避免 undefined 导致误判
    const uniqueKey = item[key];
    if (!map.has(uniqueKey)) {
      map.set(uniqueKey, true);
      return true; // 保留当前元素
    }
    return false; // 过滤重复元素
  });
}

// 测试示例
const objArr = [
  { id: 1, name: '张三', age: 20 },
  { id: 1, name: '张三', age: 21 }, // id重复，会被过滤
  { id: 2, name: '李四', age: 22 },
  { id: 3, name: '王五', age: 23 },
  { id: 2, name: '李四', age: 24 }  // id重复，会被过滤
];

// 按 id 去重
const uniqueArr = uniqueObjArrByKey(objArr, 'id');
console.log(uniqueArr);
// 输出：[{id:1,name:'张三',age:20}, {id:2,name:'李四',age:22}, {id:3,name:'王五',age:23}]
```

**2. 对象 + 遍历**

```js
function uniqueObjArrByKey(arr, key) {
  if (!Array.isArray(arr) || !key) return arr;
  
  const obj = {};
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    const item = arr[i];
    const uniqueKey = item[key];
    // 用 属性值 作为对象key，避免重复
    if (!obj[uniqueKey]) {
      obj[uniqueKey] = true;
      result.push(item);
    }
  }
  return result;
}
```

### 多个属性联合去重

```js
function uniqueObjArrByKeys(arr, keys) {
  if (!Array.isArray(arr) || !Array.isArray(keys) || keys.length === 0) return arr;
  
  const map = new Map();
  return arr.filter(item => {
    // 拼接多个属性值作为唯一标识（如 id=1&name=张三）
    const uniqueKey = keys.map(k => item[k]).join('&');
    if (!map.has(uniqueKey)) {
      map.set(uniqueKey, true);
      return true;
    }
    return false;
  });
}

// 测试示例
const objArr = [
  { id: 1, name: '张三', age: 20 },
  { id: 1, name: '张三', age: 21 }, // id+name都重复，过滤
  { id: 1, name: '李四', age: 22 }, // id相同但name不同，保留
  { id: 2, name: '张三', age: 23 }  // name相同但id不同，保留
];

// 按 id + name 联合去重
const uniqueArr = uniqueObjArrByKeys(objArr, ['id', 'name']);
console.log(uniqueArr);
// 输出：[{id:1,name:'张三',age:20}, {id:1,name:'李四',age:22}, {id:2,name:'张三',age:23}]
```

**按对象完整内容去重**
- 如果没有明确的去重属性，要求 “对象所有键值对都相同才视为重复”

```js
function uniqueObjArrByContent(arr) {
  if (!Array.isArray(arr)) return arr;
  
  const map = new Map();
  return arr.filter(item => {
    // 将对象转为JSON字符串作为唯一标识（注意：属性顺序会影响结果）
    const uniqueKey = JSON.stringify(item);
    if (!map.has(uniqueKey)) {
      map.set(uniqueKey, true);
      return true;
    }
    return false;
  });
}

// 测试示例
const objArr = [
  { a: 1, b: 2 },
  { a: 1, b: 2 }, // 内容完全相同，过滤
  { b: 2, a: 1 }, // 属性顺序不同，JSON字符串不同，保留
  { a: 1, b: 3 }  // 内容不同，保留
];

const uniqueArr = uniqueObjArrByContent(objArr);
console.log(uniqueArr);
// 输出：[{a:1,b:2}, {b:2,a:1}, {a:1,b:3}]
```

### 扩展：保留最后一次出现的元素

默认去重会保留首次出现的元素，如果需要保留最后一次出现的（比如最新数据），只需调整逻辑

```js
function uniqueObjArrKeepLast(arr, key) {
  if (!Array.isArray(arr) || !key) return arr;
  
  const map = new Map();
  // 先遍历，覆盖重复的key，保留最后一次的元素
  arr.forEach(item => {
    map.set(item[key], item);
  });
  // 把Map的值转为数组
  return Array.from(map.values());
}

// 测试
const objArr = [
  { id: 1, age: 20 },
  { id: 1, age: 21 }, // 最后一次出现，会保留这个
  { id: 2, age: 22 }
];
console.log(uniqueObjArrKeepLast(objArr, 'id'));
// 输出：[{id:1,age:21}, {id:2,age:22}]
```
